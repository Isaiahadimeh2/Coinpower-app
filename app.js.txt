async function handleWithdrawal(amount, userLevel) {
    // 1. Check Activation Rule: Must be PG2 or higher to withdraw
    if (userLevel === 'PG1') {
        alert("Withdrawal Locked: You must upgrade to at least PG2 to unlock the gateway.");
        return;
    }

    // 2. Minimum amount check
    if (amount < 10) {
        alert("Minimum withdrawal is $10.");
        return;
    }

    // 3. Calculate the 15% Global Processing Fee
    const FEE_RATE = 0.15;
    const feeAmount = amount * FEE_RATE;
    const finalAmount = amount - feeAmount;

    // 4. Submit to Supabase Transactions Table
    const { data, error } = await supabase
        .from('transactions')
        .insert([
            { 
                user_id: auth.uid(), 
                amount: amount, 
                fee: feeAmount, // The 15% maintenance tax
                status: 'pending', // Requires manual verification
                payment_details: 'Mobile Money / Bank Transfer' 
            }
        ]);

    if (error) {
        console.error("Error submitting withdrawal:", error.message);
    } else {
        alert(`Withdrawal of $${finalAmount.toFixed(2)} submitted! (Fee: $${feeAmount.toFixed(2)})`);
        // Note: Processing takes 1 to 24 hours
    }
}
